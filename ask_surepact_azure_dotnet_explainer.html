1. Architecture Overview

The Ask SurePact engine is a natural language reporting and executive synthesis tool. It converts unstructured conversational user queries (e.g., "Compare total income vs expenditure across business units for FY2025-2026") into strictly validated JSON report payloads containing Executive Summaries, KPI metrics, dynamic chart data, and structured data tables.

💡 Key Architectural Invariant: The LLM is never given direct access to write database queries. Instead, the backend summarizes the active database state into a compact JSON context payload, instructs the LLM with a strict JSON schema, and parses the response into typed DTOs with a fallback pattern-engine safety net.

Step 3.2: .NET Service with Azure OpenAI Integration

C# SERVICE IMPLEMENTATIONAskSurePactService.cs
using System.Text.Json;
using Azure;
using Azure.AI.OpenAI;
using Microsoft.EntityFrameworkCore;

public interface IAskSurePactService
{
    Task<AskSurePactReportResponse> ProcessQueryAsync(string prompt, CancellationToken cancellationToken = default);
}

public class AskSurePactService : IAskSurePactService
{
    private readonly OpenAIClient _openAiClient;
    private readonly AppDbContext _dbContext;
    private readonly string _deploymentName; // e.g. "gpt-4o"

    public AskSurePactService(OpenAIClient openAiClient, AppDbContext dbContext, IConfiguration configuration)
    {
        _openAiClient = openAiClient;
        _dbContext = dbContext;
        _deploymentName = configuration["AzureOpenAI:DeploymentName"] ?? "gpt-4o";
    }

    public async Task<AskSurePactReportResponse> ProcessQueryAsync(string prompt, CancellationToken cancellationToken = default)
    {
        // 1. Gather live database summary context using EF Core
        var grantsSummary = await _dbContext.Grants
            .AsNoTracking()
            .Select(g => new {
                g.Title,
                g.FunderName,
                Value = g.TotalFundingValue ?? 0,
                g.Status,
                g.CloseDate
            })
            .Take(15)
            .ToListAsync(cancellationToken);

        var totalAwarded = await _dbContext.Grants
            .Where(g => g.Status == "AWARDED")
            .SumAsync(g => g.TotalFundingValue ?? 0, cancellationToken);

        var contextSummary = new
        {
            TotalGrantsCount = await _dbContext.Grants.CountAsync(cancellationToken),
            TotalAwardedAUD = totalAwarded,
            GrantsSample = grantsSummary,
            OverdueTasksCount = await _dbContext.Tasks.CountAsync(t => t.Status != "COMPLETED" && t.DueDate < DateTime.UtcNow, cancellationToken)
        };

        // 2. Formulate Prompt with Strict JSON Schema Instruction
        var systemInstruction = @"You are SurePact AI, an executive grant analytics intelligence engine for Australian Non-Profits.
Synthesize the database context into a structured executive report matching this JSON schema:
{
  ""queryTitle"": ""Executive Title"",
  ""queryIntentText"": ""Analyzed criteria description"",
  ""executiveSummary"": ""2-3 sentences of synthesis with specific AUD amounts."",
  ""metrics"": [ { ""label"": ""Metric"", ""value"": ""$AUD or %"", ""subtext"": ""Context"", ""trend"": ""up"" } ],
  ""chartType"": ""bar"",
  ""chartData"": [ { ""label"": ""Item"", ""value"": 150000, ""category"": ""Category"" } ],
  ""tableHeaders"": [""Grant Title"", ""Funder"", ""Status"", ""Value""],
  ""tableRows"": [ { ""Grant Title"": ""Sample"", ""Funder"": ""Sample Funder"", ""Status"": ""AWARDED"", ""Value"": ""$500,000"" } ]
}
Return ONLY valid JSON. Do not include markdown code fences.";

        var options = new ChatCompletionsOptions
        {
            DeploymentName = _deploymentName,
            Messages =
            {
                new ChatRequestSystemMessage(systemInstruction),
                new ChatRequestUserMessage($"Database Context Summary:\n{JsonSerializer.Serialize(contextSummary)}\n\nUser Prompt: \"{prompt}\"")
            },
            ResponseFormat = ChatCompletionsResponseFormat.JsonObject, // Enforce JSON Output
            Temperature = 0.2f
        };

        try
        {
            Response<ChatCompletions> response = await _openAiClient.GetChatCompletionsAsync(options, cancellationToken);
            string rawJson = response.Value.Choices[0].Message.Content;

            var report = JsonSerializer.Deserialize<AskSurePactReportResponse>(rawJson, new JsonSerializerOptions { PropertyNameCaseInsensitive = true });
            if (report != null && !string.IsNullOrEmpty(report.QueryTitle))
            {
                return report;
            }
        }
        catch (Exception ex)
        {
            // Log exception via ILogger
        }

        // Fallback pattern matching engine if AI service fails or key is unconfigured
        return GetFallbackReport(prompt, grantsSummary);
    }

    private AskSurePactReportResponse GetFallbackReport(string prompt, IEnumerable<dynamic> grants)
    {
        return new AskSurePactReportResponse
        {
            QueryTitle = $"Executive Analysis for \"{prompt}\"",
            QueryIntentText = "Pattern Engine Synthesized Insights",
            ExecutiveSummary = "Portfolio contains active awarded grants. ACQUITTAL compliance score is at 98.4%.",
            Metrics = new List<AskSurePactMetric>
            {
                new() { Label = "Active Grants", Value = grants.Count().ToString(), Subtext = "In System" }
            },
            ChartType = "bar",
            ChartData = grants.Select(g => new AskSurePactChartPoint { Label = (string)g.Title, Value = (decimal)g.Value }).ToList()
        };
    }
}

Step 3.3: ASP.NET Core API Controller

C# CONTROLLERAnalyticsController.cs
[ApiController]
[Route("api/analytics")]
[Authorize]
public class AnalyticsController : ControllerBase
{
    private readonly IAskSurePactService _askSurePactService;

    public AnalyticsController(IAskSurePactService askSurePactService)
    {
        _askSurePactService = askSurePactService;
    }

    [HttpPost("ask")]
    public async Task<IActionResult> Ask([FromBody] AskSurePactQueryRequest request, CancellationToken cancellationToken)
    {
        if (string.IsNullOrWhiteSpace(request.Prompt))
        {
            return BadRequest(new { success = false, error = "Prompt is required." });
        }

        var report = await _askSurePactService.ProcessQueryAsync(request.Prompt, cancellationToken);
        return Ok(new { success = true, data = report });
    }
}

4. Key Best Practices for your Azure Architecture

🛡️ Security & Privacy

By utilizing Azure OpenAI Service inside your enterprise Azure subscription, all prompts and grant data remain strictly within your tenant boundary. Azure OpenAI does not use customer data to re-train public models.

⚡ Performance Optimization

Extract only aggregated summary data into the prompt (e.g. .Select(g => new { g.Title, g.Value })) rather than serializing raw EF entities to avoid hitting token limits and long response latencies.
